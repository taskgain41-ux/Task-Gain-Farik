
import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { translations } from '../translations';

const Login: React.FC<{ onBack: () => void }> = ({ onBack }) => {
  const { login, lang } = useApp();
  const t = translations[lang];
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [isAdminMode, setIsAdminMode] = useState(false);
  const [error, setError] = useState('');

  const handleLogin = (e: React.FormEvent) => {
    e.preventDefault();
    const success = login(email, isAdminMode ? password : '');
    if (success) {
      onBack();
    } else {
      setError('Invalid credentials');
    }
  };

  return (
    <div className="max-w-md mx-auto mt-12 animate-fadeIn">
      <div className="bg-white dark:bg-slate-900 rounded-3xl shadow-2xl p-8 border border-slate-100 dark:border-slate-800 relative overflow-hidden transition-colors">
        <div className="absolute top-0 left-0 w-full h-2 bg-indigo-600"></div>
        
        <div className="text-center mb-8">
          <h2 className="text-3xl font-extrabold text-slate-800 dark:text-white">{isAdminMode ? 'Admin Login' : t.login}</h2>
          <p className="text-slate-500 dark:text-slate-400 mt-2">
            {isAdminMode ? 'Manage TaskGain system' : 'Enter your email to start earning'}
          </p>
        </div>

        <form onSubmit={handleLogin} className="space-y-6">
          <div className="space-y-2">
            <label className="text-sm font-bold text-slate-700 dark:text-slate-300 px-1">Email Address</label>
            <input
              type="email"
              required
              placeholder="name@example.com"
              className="w-full px-4 py-3 rounded-xl border border-slate-200 dark:border-slate-700 dark:bg-slate-800 dark:text-white focus:ring-2 focus:ring-indigo-500 outline-none transition"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
            />
          </div>

          {isAdminMode && (
            <div className="space-y-2">
              <label className="text-sm font-bold text-slate-700 dark:text-slate-300 px-1">Admin Password</label>
              <input
                type="password"
                required
                placeholder="••••••••"
                className="w-full px-4 py-3 rounded-xl border border-slate-200 dark:border-slate-700 dark:bg-slate-800 dark:text-white focus:ring-2 focus:ring-indigo-500 outline-none transition"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
              />
            </div>
          )}

          {error && <p className="text-rose-600 dark:text-rose-400 text-sm font-medium text-center">{error}</p>}

          <button
            type="submit"
            className="w-full bg-indigo-600 text-white py-4 rounded-xl font-bold text-lg hover:bg-indigo-700 transition shadow-lg shadow-indigo-100 dark:shadow-none"
          >
            {isAdminMode ? 'Admin Portal' : t.login}
          </button>
        </form>

        <div className="mt-8 flex flex-col gap-4">
          <button
            onClick={() => setIsAdminMode(!isAdminMode)}
            className="text-indigo-600 dark:text-indigo-400 text-sm font-semibold hover:underline"
          >
            {isAdminMode ? 'Switch to User Login' : 'Login as Admin'}
          </button>
          <button
            onClick={onBack}
            className="text-slate-400 dark:text-slate-500 text-sm font-medium hover:text-slate-600 dark:hover:text-slate-300 transition-colors"
          >
            ← Back to Home
          </button>
        </div>
      </div>
    </div>
  );
};

export default Login;
